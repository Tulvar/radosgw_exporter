{{- $createSecret := eq (include "radosgw-exporter.createSecret" .) "true" }}

1. RADOSGW Exporter has been deployed!

2. Credentials configuration:
{{- if .Values.exporter.credentials.existingSecret.name }}
   ✓ Using existing secret: {{ .Values.exporter.credentials.existingSecret.name }}
{{- else if $createSecret }}
   ✓ Created new secret: {{ $secretName }}
{{- if and .Values.exporter.credentials.accessKey .Values.exporter.credentials.secretKey }}
   ✓ Credentials were provided during installation
{{- else }}
   ⚠️  Secret was created with empty credentials
   You need to update the secret with valid credentials:
   kubectl create secret generic {{ $secretName }} \
     --namespace {{ .Release.Namespace }} \
     --from-literal=accessKey=<YOUR_ACCESS_KEY> \
     --from-literal=secretKey=<YOUR_SECRET_KEY> \
     --dry-run=client -o yaml | kubectl apply -f -
{{- end }}
{{- else }}
   ✗ No credentials configured! The exporter will fail to start.
{{- end }}

3. Check if the exporter is running:
   kubectl get pods -n {{ .Release.Namespace }} \
     -l app.kubernetes.io/name={{ include "radosgw-exporter.name" . }}

4. Access metrics:
   # Port-forward to local machine
   kubectl port-forward svc/{{ include "radosgw-exporter.fullname" . }} \
     {{ .Values.service.port }}:{{ .Values.exporter.metricsPort }} \
     -n {{ .Release.Namespace }}
   
   # Then in another terminal
   curl http://localhost:{{ .Values.service.port }}/metrics

{{- if $createSecret }}
5. To update credentials later:
   kubectl create secret generic {{ $secretName }} \
     --namespace {{ .Release.Namespace }} \
     --from-literal=accessKey=<NEW_ACCESS_KEY> \
     --from-literal=secretKey=<NEW_SECRET_KEY> \
     --dry-run=client -o yaml | kubectl apply -f -
   
   # Restart the pod to pick up new credentials
   kubectl rollout restart deployment/{{ include "radosgw-exporter.fullname" . }} \
     -n {{ .Release.Namespace }}
{{- end }}
